      // Minimal Chart.js compatible stub — no CDN needed
      window.Chart = (function(){
      function Chart(canvas, config) {
      this.canvas = canvas;
      this.config = config;
      this.data = config.data || {datasets:[]};
      this._draw();
      }
      Chart.prototype._draw = function() {
      try {
      const ctx = this.canvas.getContext('2d');
      const w = this.canvas.width || this.canvas.offsetWidth || 300;
      const h = this.canvas.height || this.canvas.offsetHeight || 200;
      this.canvas.width = w; this.canvas.height = h;
      ctx.clearRect(0, 0, w, h);
      const type = this.config.type;
      const data = this.data;
      const labels = data.labels || [];
      const datasets = data.datasets || [];
      const opts = this.config.options || {};
      if (!datasets.length || !labels.length) {
      ctx.fillStyle = '#475569';
      ctx.font = '12px sans-serif';
      ctx.textAlign = 'center';
      ctx.fillText('No data', w/2, h/2);
      return;
      }
      if (type === 'doughnut' || type === 'pie') {
      this._drawDonut(ctx, w, h, datasets, labels);
      } else {
      this._drawBar(ctx, w, h, datasets, labels, type === 'line');
      }
      } catch(e) {}
      };
      Chart.prototype._drawBar = function(ctx, w, h, datasets, labels, isLine) {
      const pad = {top:20, right:10, bottom:35, left:45};
      const cw = w - pad.left - pad.right;
      const ch = h - pad.top - pad.bottom;
      // Find max value
      let maxVal = 0;
      datasets.forEach(ds => (ds.data||[]).forEach(v => { if(+v > maxVal) maxVal = +v; }));
      if (!maxVal) maxVal = 1;
      // Draw grid
      ctx.strokeStyle = 'rgba(255,255,255,0.08)';
      ctx.lineWidth = 1;
      for (let i=0;i<=4;i++) {
      const y = pad.top + ch - (ch*i/4);
      ctx.beginPath(); ctx.moveTo(pad.left, y); ctx.lineTo(pad.left+cw, y); ctx.stroke();
      ctx.fillStyle = '#64748b'; ctx.font = '9px sans-serif'; ctx.textAlign = 'right';
      ctx.fillText(Math.round(maxVal*i/4), pad.left-4, y+3);
      }
      // Draw bars/lines
      const n = labels.length;
      const bw = Math.max(4, (cw / (n * datasets.length + 1)) - 2);
      datasets.forEach((ds, di) => {
      const color = (Array.isArray(ds.backgroundColor) ? ds.backgroundColor[0] : ds.backgroundColor) || ds.borderColor || '#3B82F6';
      const pts = [];
      (ds.data||[]).forEach((val, li) => {
      const x = pad.left + (li * cw / n) + (di * (bw+2)) + bw/2 + 2;
      const y = pad.top + ch - (ch * Math.min(+val, maxVal) / maxVal);
      pts.push({x, y});
      if (!isLine) {
      ctx.fillStyle = typeof color === 'string' ? color : '#3B82F6';
      ctx.fillRect(x - bw/2, y, bw, pad.top + ch - y);
      }
      });
      if (isLine && pts.length) {
      ctx.strokeStyle = ds.borderColor || '#3B82F6';
      ctx.lineWidth = 2; ctx.beginPath();
      pts.forEach((p,i) => i===0 ? ctx.moveTo(p.x,p.y) : ctx.lineTo(p.x,p.y));
      ctx.stroke();
      pts.forEach(p => { ctx.fillStyle=ds.borderColor||'#3B82F6'; ctx.beginPath(); ctx.arc(p.x,p.y,3,0,Math.PI*2); ctx.fill(); });
      }
      });
      // X labels
      ctx.fillStyle = '#64748b'; ctx.font = '9px sans-serif'; ctx.textAlign = 'center';
      labels.forEach((lbl, li) => {
      const x = pad.left + (li * cw / n) + cw/(n*2);
      ctx.fillText(String(lbl).slice(0,3), x, pad.top + ch + 14);
      });
      };
      Chart.prototype._drawDonut = function(ctx, w, h, datasets, labels) {
      const cx=w/2, cy=h/2, r=Math.min(cx,cy)*0.7, ri=r*0.55;
      const vals = datasets[0].data || [];
      const colors = datasets[0].backgroundColor || ['#3B82F6','#10B981','#F59E0B','#EF4444','#8B5CF6','#60A5FA'];
      const total = vals.reduce((a,b)=>a+(+b||0),0) || 1;
      let angle = -Math.PI/2;
      vals.forEach((v, i) => {
      const slice = (v/total) * Math.PI * 2;
      ctx.beginPath(); ctx.moveTo(cx,cy);
      ctx.arc(cx, cy, r, angle, angle+slice);
      ctx.fillStyle = colors[i%colors.length];
      ctx.fill();
      angle += slice;
      });
      ctx.beginPath(); ctx.arc(cx,cy,ri,0,Math.PI*2);
      ctx.fillStyle = '#1e2433'; ctx.fill();
      };
      Chart.prototype.update = function() { this._draw(); };
      Chart.prototype.destroy = function() {
      try { const ctx = this.canvas.getContext('2d'); ctx.clearRect(0,0,this.canvas.width,this.canvas.height); } catch(e) {}
      };
      Chart.register = function(){};
      return Chart;
      })();
